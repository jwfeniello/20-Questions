cmake_minimum_required(VERSION 3.16)
project(twentyq VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(q20 STATIC src/model.c src/engine.c)
target_include_directories(q20 PUBLIC include)
add_executable(twentyq src/cli.c)
target_link_libraries(twentyq PRIVATE q20)
if(MSVC)
  target_compile_options(q20 PRIVATE /W4)
else()
  target_compile_options(q20 PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_library(q20_test SHARED src/model.c src/engine.c tests/bridge.c)
  target_include_directories(q20_test PRIVATE include src)
  set_target_properties(q20_test PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME original_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compare.py --library $<TARGET_FILE:q20_test>)
    add_test(NAME reader_rng_edges COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/edges.py --library $<TARGET_FILE:q20_test>)
    add_test(NAME cli_and_container COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py --exe $<TARGET_FILE:twentyq> --library $<TARGET_FILE:q20_test>)
  endif()
endif()
